cmake_minimum_required(VERSION 3.16)
project(hoiko_linux LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(hoiko_core SHARED native/bridge.cpp)
target_include_directories(hoiko_core PRIVATE native)
target_compile_options(hoiko_core PRIVATE -fno-strict-aliasing)
option(HOIKO_SANITIZE "Build core with address and undefined-behavior checks" OFF)
if(HOIKO_SANITIZE)
  target_compile_options(hoiko_core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(hoiko_core PRIVATE -fsanitize=address,undefined)
endif()
